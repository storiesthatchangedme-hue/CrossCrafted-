    33→async function resolveUserId(idOrLegacy) {
    34→  // Try UUID first (Supabase primary key)
    35→  if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(idOrLegacy)) {
    36→    const { data } = await sb.from('users').select('id, legacy_id').eq('id', idOrLegacy).single();
    37→    return data?.id || null;
    38→  }
    39→  // Try legacy_id
    40→  const { data } = await sb.from('users').select('id, legacy_id').eq('legacy_id', idOrLegacy).single();
    41→  return data?.id || null;
    42→}
    43→
    44→async function resolveId(table, idOrLegacy) {
    45→  if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(idOrLegacy)) {
    46→    return idOrLegacy;
    47→  }
    48→  const { data } = await sb.from(table).select('id').eq('legacy_id', idOrLegacy).single();
    49→  return data?.id || idOrLegacy;
    50→}
    51→
    52→function generateLegacyId(prefix) {
    53→  return `${prefix}_${uuidv4().slice(0, 8)}`;
    54→}

    77→  async findAll(opts = {}) {
    78→    let query = sb.from('users').select('*');
    79→    if (opts.role) query = query.eq('role', opts.role);
    80→    if (opts.status) query = query.eq('status', opts.status);
    81→    if (opts.search) {
    82→      query = query.or(`name.ilike.%${opts.search}%,username.ilike.%${opts.search}%,email.ilike.%${opts.search}%`);
    83→    }
    84→    if (opts.limit) query = query.limit(opts.limit);
    85→    if (opts.offset) query = query.range(opts.offset, opts.offset + (opts.limit || 20) - 1);
    86→    const { data } = await query.order('created_at', { ascending: false });
    87→    return data || [];
    88→  },

    90→  async create(userData) {
    91→    const id = userData.id || uuidv4();
    92→    const legacyId = userData.legacy_id || userData._id || generateLegacyId('user');
    93→    const { password_hash, _id, ...safeFields } = userData;
    94→
    95→    const insertData = {
    96→      id,
    97→      legacy_id: legacyId,
    98→      ...safeFields,
    99→    };
   100→
   101→    // Map old field names to new schema
   102→    if (insertData.faith_belief) insertData.denomination = insertData.faith_belief;
   103→    if (insertData.faith_journey) insertData.faith_journey_status = insertData.faith_journey;
   104→
   105→    const { data, error } = await sb.from('users').insert(insertData).select('*').single();
   106→    if (error) throw error;
   107→    return { ...data, _id: data.id || legacyId, legacy_id: legacyId };
   108→  },

   110→  async update(id, updates) {

   115→    return { ...data, _id: data.id || data.legacy_id };

   183→  async toggleLike(postId, userId) {
   184→    const resolvedId = await resolveId('posts', postId);
   185→    const post = await this.findById(resolvedId);
   186→    if (!post) return null;
   187→
   188→    const likes = post.likes || [];
   189→    const idx = likes.indexOf(userId);
   190→
   191→    if (idx > -1) {
   192→      likes.splice(idx, 1);
   193→    } else {
   194→      likes.push(userId);
   195→    }
   196→
   197→    const { data, error } = await sb.from('posts').update({ likes }).eq('id', resolvedId).select('*').single();
   198→    if (error) throw error;

   202→  async savePost(postId, userId) {
   203→    const resolvedId = await resolveId('users', userId);
   204→    const user = await users.findById(resolvedId);
   205→    if (!user) return;
   206→
   207→    const saved = user.saved_posts || [];
   208→    if (!saved.includes(postId)) {
   209→      saved.push(postId);
   210→      await users.update(resolvedId, { saved_posts: saved });
   211→    }
   212→  },
   213→
   214→  async unsavePost(postId, userId) {
   215→    const resolvedId = await resolveId('users', userId);
   216→    const user = await users.findById(resolvedId);
   217→    if (!user) return;
   218→
   219→    const saved = (user.saved_posts || []).filter(id => id !== postId);
   220→    await users.update(resolvedId, { saved_posts: saved });
   221→  },
   222→
   223→  async getSavedPosts(userId) {
   224→    const user = await users.findById(userId);
   225→    if (!user) return [];
   226→    const savedIds = user.saved_posts || [];
   227→    if (savedIds.length === 0) return [];
   228→
   229→    const allPosts = await this.findAll(userId);
   230→    return allPosts.filter(p => savedIds.includes(p.id) || savedIds.includes(p._id));
   231→  },

   233→  async getByUser(userId) {
   234→    const { data } = await sb.from('posts').select('*').eq('user_id', userId).order('created_at', { ascending: false });
   235→    return (data || []).map(p => ({ ...p, _id: p.id || p.legacy_id }));

   242→    if (error) throw error;

   680→  async findByUser(userId) {
   681→    const { data: participants } = await sb.from('conversation_participants')
   682→      .select('conversation_id')
   683→      .eq('user_id', userId);
   684→
   685→    if (!participants || participants.length === 0) return [];
   686→
   687→    const convoIds = participants.map(p => p.conversation_id);
   688→    const results = [];
   689→
   690→    for (const convoId of convoIds) {
   691→      const { data: parts } = await sb.from('conversation_participants')
   692→        .select('user_id')
   693→        .eq('conversation_id', convoId);
   694→
   695→      const otherUserId = parts?.find(p => p.user_id !== userId)?.user_id;
   696→      if (!otherUserId) continue;
   697→
   698→      const other = await users.findById(otherUserId);
   699→      const { data: lastMsg } = await sb.from('messages')
   700→        .select('*')
   701→        .eq('conversation_id', convoId)
   702→        .order('created_at', { ascending: false })
   703→        .limit(1)
   704→        .single();
   705→
   706→      const unreadCount = await sb.from('messages')
   707→        .select('*', { count: 'exact', head: true })
   708→        .eq('conversation_id', convoId)
   709→        .eq('is_read', false)
   710→        .neq('sender_id', userId);
   711→
   712→      results.push({
   713→        _id: convoId,
   714→        other_user: other ? {
   715→          _id: other.id,
   716→          name: other.name,
   717→          username: other.username,
   718→          profile_image: other.profile_image,
   719→          is_verified: other.is_verified,
   720→        } : null,
   721→        last_message: lastMsg?.content || '',
   722→        last_message_at: lastMsg?.created_at || '',
   723→        unread_count: unreadCount.count || 0,
   724→      });
   725→    }
   726→
   727→    results.sort((a, b) => new Date(b.last_message_at) - new Date(a.last_message_at));
   728→    return results;
   729→  },

   971→const bff = {
   972→  async swipe(swiperId, swipedId, direction) {
   973→    const { error } = await sb.from('bff_swipes').insert({
   974→      swiper_id: swiperId,
   975→      swiped_id: swipedId,
   976→      direction,
   977→    });
   978→    if (error) console.error('[DB] Swipe error:', error.message);
   979→
   980→    // Check for mutual like
   981→    if (direction === 'right') {
   982→      const { data } = await sb.from('bff_swipes').select('id')
   983→        .eq('swiper_id', swipedId)
   984→        .eq('swiped_id', swiperId)
   985→        .eq('direction', 'right')
   986→        .single();
   987→
   988→      if (data) {
   989→        await sb.from('bff_connections').insert({
   990→          user1_id: swiperId,
   991→          user2_id: swipedId,
   992→        });
   993→        return { matched: true };
   994→      }
   995→    }
   996→    return { matched: false };
   997→  },
   998→
   999→  async getConnections(userId) {
  1000→    const { data } = await sb.from('bff_connections').select('*')
  1001→      .or(`user1_id.eq.${userId},user2_id.eq.${userId}`)
  1002→      .order('created_at', { ascending: false });
  1003→
  1004→    return (data || []).map(c => ({
  1005→      ...c,
  1006→      _id: c.id,
  1007→      other_user_id: c.user1_id === userId ? c.user2_id : c.user1_id,
  1008→    }));
  1009→  },

  1011→  async getDiscoverUsers(userId, limit = 10) {
  1012→    // Get users not yet swiped on, excluding self and blocked
  1013→    const { data: swiped } = await sb.from('bff_swipes').select('swiped_id')
  1014→      .eq('swiper_id', userId);
  1015→
  1016→    const swipedIds = (swiped || []).map(s => s.swiped_id);
  1017→    swipedIds.push(userId);
  1018→
  1019→    const { data } = await sb.from('users').select('id, name, username, profile_image, bio, city, state, denomination, interests, looking_for, age, gender')
  1020→      .not('id', 'in', `(${swipedIds.join(',')})`)
  1021→    .neq('status', 'rejected')
  1022→    .limit(limit);
  1023→
  1024→    return (data || []).map(u => ({ ...u, _id: u.id }));
  1025→  }

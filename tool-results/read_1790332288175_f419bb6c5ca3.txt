   160→def hash_password(password: str) -> str:
   161→    salt = bcrypt.gensalt()
   162→    hashed = bcrypt.hashpw(password.encode("utf-8"), salt)
   163→    return hashed.decode("utf-8")
   164→
   165→def verify_password(plain_password: str, hashed_password: str) -> bool:
   166→    return bcrypt.checkpw(plain_password.encode("utf-8"), hashed_password.encode("utf-8"))

   178→async def get_current_user(request: Request) -> dict:
   179→    # Try session_token first (Google OAuth)
   180→    session_token = request.cookies.get("session_token")
   181→    if not session_token:
   182→        auth_header = request.headers.get("Authorization", "")
   183→        if auth_header.startswith("Bearer "):
   184→            session_token = auth_header[7:]
   185→
   186→    if session_token:
   187→        session = await db.user_sessions.find_one({"session_token": session_token})
   188→        if session:
   189→            expires_at = session.get("expires_at", datetime.min)
   190→            if isinstance(expires_at, str):
   191→                expires_at = datetime.fromisoformat(expires_at)
   192→            if expires_at.tzinfo is None:
   193→                expires_at = expires_at.replace(tzinfo=timezone.utc)
   194→            if expires_at > datetime.now(timezone.utc):
   195→                user = await db.users.find_one({"_id": ObjectId(session["user_id"])})
   196→                if user:
   197→                    user["_id"] = str(user["_id"])
   198→                    user.pop("password_hash", None)
   199→                    return user
   200→
   201→    # Try JWT access_token (email/password auth)
   202→    token = request.cookies.get("access_token")
   203→    if not token:
   204→        auth_header = request.headers.get("Authorization", "")
   205→        if auth_header.startswith("Bearer "):
   206→            token = auth_header[7:]
   207→    if not token:
   208→        raise HTTPException(status_code=401, detail="Not authenticated")
   209→    try:
   210→        payload = jwt.decode(token, get_jwt_secret(), algorithms=[JWT_ALGORITHM])
   211→        if payload.get("type") != "access":
   212→            raise HTTPException(status_code=401, detail="Invalid token type")
   213→        user = await db.users.find_one({"_id": ObjectId(payload["sub"])})
   214→        if not user:
   215→            raise HTTPException(status_code=401, detail="User not found")
   216→        user["_id"] = str(user["_id"])
   217→        user.pop("password_hash", None)
   218→        return user
   219→    except jwt.ExpiredSignatureError:
   220→        raise HTTPException(status_code=401, detail="Token expired")
   221→    except jwt.InvalidTokenError:
   222→        raise HTTPException(status_code=401, detail="Invalid token")

   401→@api_router.post("/auth/login")
   402→async def login(request: LoginRequest, response: Response, req: Request):
   403→    email = request.email.lower()
   404→    identifier = f"{req.client.host}:{email}"
   405→    
   406→    attempt = await db.login_attempts.find_one({"identifier": identifier})
   407→    if attempt and attempt.get("locked_until"):
   408→        locked = attempt["locked_until"]
   409→        if locked.tzinfo is None:
   410→            locked = locked.replace(tzinfo=timezone.utc)
   411→        if locked > datetime.now(timezone.utc):
   412→            raise HTTPException(status_code=429, detail="Too many failed attempts. Try again later.")
   413→    
   414→    user = await db.users.find_one({"email": email})
   415→    if not user or not verify_password(request.password, user["password_hash"]):
   416→        await db.login_attempts.update_one(
   417→            {"identifier": identifier},
   418→            {"$inc": {"count": 1}, "$set": {"last_attempt": datetime.now(timezone.utc)}},
   419→            upsert=True
   420→        )
   421→        current_attempt = await db.login_attempts.find_one({"identifier": identifier})
   422→        if current_attempt and current_attempt.get("count", 0) >= 5:
   423→            await db.login_attempts.update_one(
   424→                {"identifier": identifier},
   425→                {"$set": {"locked_until": datetime.now(timezone.utc) + timedelta(minutes=15)}}
   426→            )
   427→        raise HTTPException(status_code=401, detail="Invalid credentials")
   428→    
   429→    await db.login_attempts.delete_one({"identifier": identifier})
   430→    
   431→    user_id = str(user["_id"])
   432→    access_token = create_access_token(user_id, email)
   433→    refresh_token = create_refresh_token(user_id)
   434→    
   435→    response.set_cookie(key="access_token", value=access_token, httponly=True, secure=False, samesite="lax", max_age=900, path="/")
   436→    response.set_cookie(key="refresh_token", value=refresh_token, httponly=True, secure=False, samesite="lax", max_age=604800, path="/")
   437→    
   438→    user["_id"] = user_id
   439→    user.pop("password_hash")
   440→    return user

   456→@api_router.post("/auth/refresh")
   457→async def refresh(request: Request, response: Response):
   458→    token = request.cookies.get("refresh_token")
   459→    if not token:
   460→        raise HTTPException(status_code=401, detail="No refresh token")
   461→    try:
   462→        payload = jwt.decode(token, get_jwt_secret(), algorithms=[JWT_ALGORITHM])
   463→        if payload.get("type") != "refresh":
   464→            raise HTTPException(status_code=401, detail="Invalid token type")
   465→        user = await db.users.find_one({"_id": ObjectId(payload["sub"])})
   466→        if not user:

   616→

   625→    if is_video:
   626→        try:

   648→async def generate_video_thumbnail(video_data: bytes, user_id: str, file_id: str) -> str:
   649→    """Extract a frame from video at 1s using ffmpeg and upload as thumbnail."""
   650→    with tempfile.TemporaryDirectory() as tmpdir:
   651→        video_path = os.path.join(tmpdir, "input.mp4")
   652→        thumb_path = os.path.join(tmpdir, "thumb.jpg")
   653→        with open(video_path, "wb") as f:
   654→            f.write(video_data)
   655→
   656→        proc = await asyncio.to_thread(
   657→            subprocess.run,
   658→            ["ffmpeg", "-i", video_path, "-ss", "00:00:01", "-vframes", "1", "-vf", "scale=640:-1", "-q:v", "4", thumb_path, "-y"],
   659→            capture_output=True, timeout=30
   660→        )
   661→
   662→        if proc.returncode != 0 or not os.path.exists(thumb_path):
   663→            # Try frame at 0s if 1s fails (short video)
   664→            proc = await asyncio.to_thread(
   665→                subprocess.run,
   666→                ["ffmpeg", "-i", video_path, "-vframes", "1", "-vf", "scale=640:-1", "-q:v", "4", thumb_path, "-y"],
   667→                capture_output=True, timeout=30
   668→            )
   669→
   670→        if os.path.exists(thumb_path):
   671→            with open(thumb_path, "rb") as f:
   672→                thumb_data = f.read()
   673→            thumb_storage_path = f"{APP_NAME}/thumbnails/{user_id}/{file_id}.jpg"
   674→            put_object(thumb_storage_path, thumb_data, "image/jpeg")
   675→            return f"/api/files/{thumb_storage_path}"
   676→
   677→    return ""

   723→        content=data,

   725→        headers={
   726→            "Accept-Ranges": "bytes",
   727→            "Content-Length": str(total_size),

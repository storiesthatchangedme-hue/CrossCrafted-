     1→import { useState, useEffect, useCallback } from 'react';
     2→import { useAuth } from '@/contexts/AuthContext';
     3→import api from '@/lib/api';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Heart, MessageCircle, Plus, Search, Award, Sparkles, Send,
     7→  Globe, Lock, BookOpen, HeartHandshake, X, ChevronDown, Check, ShieldCheck
     8→} from 'lucide-react';
     9→import { toast } from 'sonner';
    10→import { PRAYER_CATEGORIES as CATEGORIES } from '@/lib/constants';
    11→
    12→
    13→
    14→
    15→const DAILY_VERSES = [
    16→  {
    17→    verse: "Do not be anxious about anything, but in every situation, by prayer and petition, with thanksgiving, present your requests to God.",
    18→    reference: "Philippians 4:6",
    19→    reflection: "No matter how big or small your request, God cares for you. Sharing your heart with Him and your brothers and sisters in Christ brings peace that transcends understanding."
    20→  },
    21→  {
    22→    verse: "Therefore confess your sins to each other and pray for each other so that you may be healed. The prayer of a righteous person is powerful and effective.",
    23→    reference: "James 5:16",
    24→    reflection: "Fellowship isn't just about sharing good times; it's about lifting each other's burdens. When we pray together, God moves in extraordinary ways."
    25→  },
    26→  {
    27→    verse: "For where two or three gather in my name, there am I with them.",
    28→    reference: "Matthew 18:20",
    29→    reflection: "We are never alone in our faith. Even online, when we unite our spirits in prayer, Jesus Christ is right here in our midst, coordinating grace."
    30→  },
    31→  {
    32→    verse: "Be joyful in hope, patient in affliction, faithful in prayer.",
    33→    reference: "Romans 12:12",
    34→    reflection: "Prayer is not a last resort, but our first response. Maintain steady faith and support each other through seasons of waiting and transformation."
    35→  }
    36→];
    37→
    38→export default function PrayerWall() {
    39→  const { user, setUser } = useAuth();
    40→  const [prayers, setPrayers] = useState([]);
    41→  const [loading, setLoading] = useState(true);
    42→  const [activeCategory, setActiveCategory] = useState('All');
    43→  const [searchQuery, setSearchQuery] = useState('');
    44→  
    45→  // Daily verse rotation
    46→  const [dailyVerse, setDailyVerse] = useState(DAILY_VERSES[0]);
    47→
    48→  // Create Modal State
    49→  const [showCreateModal, setShowCreateModal] = useState(false);
    50→  const [newTitle, setNewTitle] = useState('');
    51→  const [newContent, setNewContent] = useState('');
    52→  const [newCategory, setNewCategory] = useState('Healing');
    53→  const [isAnonymous, setIsAnonymous] = useState(false);
    54→  const [submitting, setSubmitting] = useState(false);
    55→
    56→  // Expanded comment states (keyed by prayer ID)
    57→  const [expandedComments, setExpandedComments] = useState({});
    58→  const [commentText, setCommentText] = useState({});
    59→
    60→  useEffect(() => {
    61→    fetchPrayers();
    62→    // Rotate verse daily/on load randomly
    63→    const idx = Math.floor(Math.random() * DAILY_VERSES.length);
    64→    setDailyVerse(DAILY_VERSES[idx]);
    65→  }, []);
    66→
    67→  const fetchPrayers = async () => {
    68→    try {
    69→      setLoading(true);
    70→      const { data } = await api.get('/api/prayers');
    71→      setPrayers(Array.isArray(data) ? data : []);
    72→    } catch (_) {
    73→      toast.error('Failed to load prayer requests');
    74→    } finally {
    75→      setLoading(false);
    76→    }
    77→  };
    78→
    79→  const handleCreatePrayer = async (e) => {
    80→    e.preventDefault();
    81→    if (!newTitle.trim() || !newContent.trim()) {
    82→      toast.error('Please enter a title and description for your prayer request');
    83→      return;
    84→    }
    85→    setSubmitting(true);
    86→
    87→    try {
    88→      const { data } = await api.post('/api/prayers', {
    89→        title: newTitle.trim(),
    90→        content: newContent.trim(),
    91→        category: newCategory,
    92→        is_anonymous: isAnonymous
    93→      });
    94→
    95→      toast.success('Prayer request shared! +10 Faith Points!');
    96→      
    97→      // Update local state
    98→      setPrayers(prev => [data, ...prev]);
    99→      
   100→      // Update user points dynamically
   101→      if (data.user_points !== undefined && setUser) {
   102→        setUser(prev => prev ? { ...prev, trivia_points: data.user_points } : prev);
   103→      }
   104→
   105→      // Reset form
   106→      setNewTitle('');
   107→      setNewContent('');
   108→      setNewCategory('Healing');
   109→      setIsAnonymous(false);
   110→      setShowCreateModal(false);
   111→    } catch (err) {
   112→      toast.error(err.response?.data?.detail || 'Failed to submit prayer request');
   113→    } finally {
   114→      setSubmitting(false);
   115→    }
   116→  };
   117→
   118→  const handlePrayToggle = async (prayerId) => {
   119→    try {
   120→      const { data } = await api.post(`/api/prayers/${prayerId}/pray`, {});
   121→      
   122→      // Feedback toast
   123→      if (data.is_praying) {
   124→        toast.success('Amen! Thank you for lifting your sibling in prayer! +5 Faith Points!');
   125→      }
   126→
   127→      // Update prayers in list
   128→      setPrayers(prev => prev.map(p => {
   129→        if (p._id === prayerId) {
   130→          return {
   131→            ...p,
   132→            praying_count: data.praying_count,
   133→            is_praying: data.is_praying
   134→          };
   135→        }
   136→        return p;
   137→      }));
   138→
   139→      // Update user points
   140→      if (data.user_points !== undefined && setUser) {
   141→        setUser(prev => prev ? { ...prev, trivia_points: data.user_points } : prev);
   142→      }
   143→    } catch (_) {
   144→      toast.error('Failed to register your prayer');
   145→    }
   146→  };
   147→
   148→  const toggleComments = (prayerId) => {
   149→    setExpandedComments(prev => ({
   150→      ...prev,
   151→      [prayerId]: !prev[prayerId]
   152→    }));
   153→  };
   154→
   155→  const handleSendComment = async (prayerId) => {
   156→    const text = commentText[prayerId] || '';
   157→    if (!text.trim()) return;
   158→
   159→    try {
   160→      const { data } = await api.post(`/api/prayers/${prayerId}/comments`, {
   161→        text: text.trim()
   162→      });
   163→
   164→      toast.success('Word of encouragement posted!');
   165→      
   166→      // Update comment list in local prayers state
   167→      setPrayers(prev => prev.map(p => {
   168→        if (p._id === prayerId) {
   169→          return { ...p, comments: data };
   170→        }
   171→        return p;
   172→      }));
   173→
   174→      // Clear input
   175→      setCommentText(prev => ({
   176→        ...prev,
   177→        [prayerId]: ''
   178→      }));
   179→    } catch (_) {
   180→      toast.error('Failed to post encouragement');
   181→    }
   182→  };
   183→
   184→  const filteredPrayers = (Array.isArray(prayers) ? prayers : []).filter(p => {
   185→    const matchesCategory = activeCategory === 'All' || p.category === activeCategory;
   186→    const matchesSearch = searchQuery.trim() === '' || 
   187→      p.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
   188→      p.content.toLowerCase().includes(searchQuery.toLowerCase()) ||
   189→      (p.user_name && p.user_name.toLowerCase().includes(searchQuery.toLowerCase()));
   190→    return matchesCategory && matchesSearch;
   191→  });
   192→
   193→  // Calculate high-level stats for active intercessors
   194→  const totalIntercessions = prayers.reduce((acc, p) => acc + (p.praying_count || 0), 0);
   195→
   196→  const getPointsLabel = (points) => {
   197→    if (points >= 300) return 'Theology Scholar';
   198→    if (points >= 150) return 'Bible Teacher';
   199→    if (points >= 50) return 'Spiritual Disciple';
   200→    return 'Faith Seeker';
   201→  };
   202→
   203→  return (
   204→    <div className="max-w-4xl mx-auto px-4 py-6" id="prayer-wall-page">
   205→      {/* ── Page Header & Navigation ── */}
   206→      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-6">
   207→        <div>
   208→          <h1 className="text-2xl md:text-3xl font-black text-white flex items-center gap-2 tracking-tight">
   209→            <HeartHandshake className="text-[#FB7185]" size={28} />
   210→            Prayer <span className="text-[#FB7185]">Wall</span>
   211→          </h1>
   212→          <p className="text-xs md:text-sm text-[#94A3B8] mt-1 font-medium">
   213→            Lifting one another up in faith, hope, and love. Intercede for fellow believers.
   214→          </p>
   215→        </div>
   216→        <button
   217→          onClick={() => setShowCreateModal(true)}
   218→          className="flex items-center justify-center gap-2 bg-gradient-to-r from-[#FB7185] to-[#E11D48] hover:scale-[1.02] active:scale-[0.98] text-white font-black text-sm px-6 py-3 rounded-2xl shadow-[0_4px_15px_rgba(251,113,133,0.3)] hover:shadow-[0_6px_20px_rgba(251,113,133,0.4)] transition-all"
   219→          id="btn-request-prayer"
   220→        >
   221→          <Plus size={18} strokeWidth={2.5} />
   222→          Request Prayer
   223→        </button>
   224→      </div>
   225→
   226→      {/* ── Daily Bread Scripture Banner ── */}
   227→      <motion.div
   228→        initial={{ opacity: 0, y: -10 }}
   229→        animate={{ opacity: 1, y: 0 }}
   230→        className="mb-6 bg-[#111827] border border-[#FB7185]/20 p-5 rounded-3xl relative overflow-hidden shadow-[0_4px_20px_rgba(0,0,0,0.2)]"
   231→        id="daily-devotional-card"
   232→      >
   233→        <div className="absolute right-0 top-0 opacity-10 blur-2xl w-32 h-32 bg-[#FB7185]/30 rounded-full pointer-events-none" />
   234→        <div className="absolute right-3 top-3 opacity-10">
   235→          <BookOpen size={96} className="text-[#FB7185]" />
   236→        </div>
   237→        <div className="flex items-center gap-1.5 text-[10px] uppercase font-black tracking-widest text-[#FB7185] mb-2 relative z-10">
   238→          <Sparkles size={12} />
   239→          Daily Devotional & Scripture
   240→        </div>
   241→        <blockquote className="text-sm md:text-base text-white/95 font-medium leading-relaxed italic pr-8 relative z-10">
   242→          "{dailyVerse.verse}"
   243→        </blockquote>
   244→        <cite className="block text-xs text-[#FB7185] font-bold tracking-wide mt-2 not-italic relative z-10">
   245→          — {dailyVerse.reference}
   246→        </cite>
   247→        <p className="text-xs text-[#64748B] mt-2 border-t border-white/[0.04] pt-2 leading-relaxed">
   248→          {dailyVerse.reflection}
   249→        </p>
   250→      </motion.div>
   251→
   252→      {/* ── Community Stat Dashboard ── */}
   253→      <div className="grid grid-cols-3 gap-3 mb-6 bg-[#111827] border border-white/[0.08] p-3.5 rounded-2xl shadow-[0_4px_15px_rgba(0,0,0,0.2)]" id="prayer-stat-dashboard">
   254→        <div className="text-center p-2 border-r border-white/[0.08]">
   255→          <span className="text-[10px] uppercase font-black tracking-widest text-[#64748B] block">Intercessions</span>
   256→          <span className="text-lg font-black text-[#FB7185] block mt-0.5">{totalIntercessions + 84}</span>
   257→          <span className="text-[8px] font-bold text-[#475569] block">Total Prayers Lifted</span>
   258→        </div>
   259→        <div className="text-center p-2 border-r border-white/[0.08]">
   260→          <span className="text-[10px] uppercase font-black tracking-widest text-[#64748B] block">Spiritual Rank</span>
   261→          <span className="text-xs font-black text-white block mt-1 truncate drop-shadow-md">
   262→            {getPointsLabel(user?.trivia_points || 0)}
   263→          </span>
   264→          <span className="text-[8px] font-bold text-[#475569] block">Based on faith points</span>
   265→        </div>
   266→        <div className="text-center p-2">
   267→          <span className="text-[10px] uppercase font-black tracking-widest text-[#64748B] block">Faith Points</span>
   268→          <span className="text-lg font-black text-[#38BDF8] block mt-0.5 drop-shadow-[0_0_8px_rgba(56,189,248,0.4)]">{user?.trivia_points || 0} XP</span>
   269→          <span className="text-[8px] font-bold text-[#475569] block">Gain from posts & prayers</span>
   270→        </div>
   271→      </div>
   272→
   273→      {/* ── Filter / Search Controls ── */}
   274→      <div className="flex flex-col gap-3.5 mb-6" id="prayer-filters-section">
   275→        {/* Search */}
   276→        <div className="relative">
   277→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 text-[#475569]" size={16} />
   278→          <input
   279→            type="text"
   280→            placeholder="Search prayers, scriptures, or names..."
   281→            value={searchQuery}
   282→            onChange={(e) => setSearchQuery(e.target.value)}
   283→            className="w-full bg-[#0F172A] border border-white/[0.06] text-white rounded-2xl pl-11 pr-4 py-2.5 text-xs focus:outline-none focus:border-amber-400/50 transition-colors"
   284→          />
   285→          {searchQuery && (
   286→            <button onClick={() => setSearchQuery('')} className="absolute right-4 top-1/2 -translate-y-1/2 text-white/40 hover:text-white">
   287→              <X size={14} />
   288→            </button>
   289→          )}
   290→        </div>
   291→
   292→        {/* Categories Pills */}
   293→        <div className="flex gap-2 overflow-x-auto pb-1.5 scrollbar-none select-none">
   294→          <button
   295→            onClick={() => setActiveCategory('All')}
   296→            className={`flex-shrink-0 px-5 py-2 rounded-full border text-xs font-black tracking-wide transition-all shadow-sm ${
   297→              activeCategory === 'All'
   298→                ? 'bg-[#FB7185] border-[#FB7185] text-slate-900 shadow-[0_4px_15px_rgba(251,113,133,0.3)]'
   299→                : 'bg-[#111827] border-white/[0.08] text-[#94A3B8] hover:text-white hover:bg-[#1F2937]'
   300→            }`}
   301→          >
   302→            All Requests
   303→          </button>
   304→          {CATEGORIES.map((cat) => (
   305→            <button
   306→              key={cat.value}
   307→              onClick={() => setActiveCategory(cat.value)}
   308→              className={`flex-shrink-0 px-5 py-2 rounded-full border text-xs font-black tracking-wide transition-all shadow-sm ${
   309→                activeCategory === cat.value
   310→                  ? 'bg-[#FB7185] border-[#FB7185] text-slate-900 shadow-[0_4px_15px_rgba(251,113,133,0.3)]'
   311→                  : 'bg-[#111827] border-white/[0.08] text-[#94A3B8] hover:text-white hover:bg-[#1F2937]'
   312→              }`}
   313→            >
   314→              {cat.label}
   315→            </button>
   316→          ))}
   317→        </div>
   318→      </div>
   319→
   320→      {/* ── Requests List ── */}
   321→      {loading ? (
   322→        <div className="flex flex-col items-center justify-center py-24 gap-3">
   323→          <div className="w-10 h-10 border-4 border-[#FB7185] border-t-transparent rounded-full animate-spin shadow-[0_0_15px_rgba(251,113,133,0.4)]"></div>
   324→          <p className="text-xs text-[#94A3B8] font-black uppercase tracking-widest mt-2">Unveiling Prayer Requests...</p>
   325→        </div>
   326→      ) : filteredPrayers.length === 0 ? (
   327→        <div className="bg-[#111827] border border-white/[0.08] rounded-3xl p-12 text-center shadow-lg">
   328→          <div className="w-16 h-16 rounded-2xl bg-[#FB7185]/10 flex items-center justify-center text-[#FB7185] border border-[#FB7185]/20 mx-auto mb-4 shadow-inner">
   329→            <HeartHandshake size={28} />
   330→          </div>
   331→          <h3 className="text-base font-black text-white mb-1">No requests found</h3>
   332→          <p className="text-xs text-[#64748B] font-medium max-w-sm mx-auto leading-relaxed">
   333→            {searchQuery 
   334→              ? "We couldn't find any prayer requests matching your terms. Try a different keyword or category."
   335→              : "Be the first to share your prayer request with the family and invite others to stand in faith with you!"}
   336→          </p>
   337→          {!searchQuery && (
   338→            <button
   339→              onClick={() => setShowCreateModal(true)}
   340→              className="mt-6 text-xs font-black uppercase tracking-widest text-[#FB7185] hover:text-[#E11D48] hover:underline transition-colors"
   341→            >
   342→              Share Request Now
   343→            </button>
   344→          )}
   345→        </div>
   346→      ) : (
   347→        <div className="space-y-4">
   348→          <AnimatePresence>
   349→            {filteredPrayers.map((prayer) => {
   350→              const isCommentExpanded = !!expandedComments[prayer._id];
   351→              const categoryDetails = CATEGORIES.find(c => c.value === prayer.category) || CATEGORIES[0];
   352→              const authorLetter = prayer.user_name ? prayer.user_name.charAt(0).toUpperCase() : 'A';
   353→              const dateString = prayer.created_at ? new Date(prayer.created_at).toLocaleDateString(undefined, {
   354→                month: 'short',
   355→                day: 'numeric',
   356→                hour: '2-digit',
   357→                minute: '2-digit'
   358→              }) : 'Just now';
   359→
   360→              return (
   361→                <motion.div
   362→                  key={prayer._id}
   363→                  layout
   364→                  initial={{ opacity: 0, y: 15 }}
   365→                  animate={{ opacity: 1, y: 0 }}
   366→                  exit={{ opacity: 0, scale: 0.95 }}
   367→                  className="bg-[#111827] border border-white/[0.08] rounded-3xl p-5 hover:border-white/[0.12] transition-all shadow-[0_4px_20px_rgba(0,0,0,0.15)] group relative overflow-hidden"
   368→                  data-testid="prayer-card"
   369→                >
   370→                  <div className="absolute top-0 right-0 w-32 h-32 bg-[#FB7185]/5 blur-3xl rounded-full pointer-events-none group-hover:bg-[#FB7185]/10 transition-colors" />
   371→                  
   372→                  {/* Card Header */}
   373→                  <div className="flex items-center justify-between gap-4 mb-4 relative z-10">
   374→                    <div className="flex items-center gap-3">
   375→                      <div className="w-12 h-12 rounded-xl overflow-hidden border-2 border-white/[0.08] bg-slate-900 flex items-center justify-center font-black text-[#FB7185] shadow-inner group-hover:scale-105 transition-transform">
   376→                        {prayer.user_image ? (
   377→                          <img src={prayer.user_image} alt="" className="w-full h-full object-cover" />
   378→                        ) : (
   379→                          <span>{authorLetter}</span>
   380→                        )}
   381→                      </div>
   382→                      <div>
   383→                        <div className="flex items-center gap-1.5">
   384→                          <span className="text-xs font-black text-white hover:text-[#FB7185] transition-colors cursor-pointer">
   385→                            {prayer.user_name}
   386→                          </span>
   387→                          {prayer.is_verified && (
   388→                            <span className="text-[#38BDF8] text-[9px] bg-[#38BDF8]/10 px-1.5 py-0.5 rounded-md uppercase font-black tracking-widest flex items-center gap-1" title="Verified Believer">
   389→                              <ShieldCheck size={10} /> Verified
   390→                            </span>
   391→                          )}
   392→                          {prayer.user_username === 'anonymous' && (
   393→                            <Lock size={12} className="text-[#64748B]" title="Private Request" />
   394→                          )}
   395→                        </div>
   396→                        <span className="text-[10px] text-[#94A3B8] font-bold block mt-0.5">{dateString}</span>
   397→                      </div>
   398→                    </div>
   399→
   400→                    <span className={`text-[9px] font-black uppercase tracking-widest px-2.5 py-1 rounded-lg border ${categoryDetails.color}`}>
   401→                      {prayer.category}
   402→                    </span>
   403→                  </div>
   404→
   405→                  {/* Card Body */}
   406→                  <div className="mb-4 relative z-10">
   407→                    <h3 className="text-[15px] font-black text-white leading-tight mb-2 group-hover:text-[#FB7185] transition-colors">{prayer.title}</h3>
   408→                    <p className="text-xs text-[#94A3B8] font-medium leading-relaxed whitespace-pre-wrap">{prayer.content}</p>
   409→                  </div>
   410→
   411→                  {/* Card Actions */}
   412→                  <div className="flex items-center gap-4 pt-4 border-t border-white/[0.08] text-xs relative z-10">
   413→                    <button
   414→                      onClick={() => handlePrayToggle(prayer._id)}
   415→                      className={`flex items-center gap-2 font-black transition-all px-4 py-2 rounded-xl border ${
   416→                        prayer.is_praying
   417→                          ? 'bg-[#FB7185]/10 text-[#FB7185] border-[#FB7185]/30 shadow-[inset_0_2px_10px_rgba(251,113,133,0.1)]'
   418→                          : 'text-[#94A3B8] hover:text-[#FB7185] hover:bg-white/[0.02] border-transparent hover:border-white/[0.05]'
   419→                      }`}
   420→                      data-testid="pray-button"
   421→                    >
   422→                      <HeartHandshake size={16} className={prayer.is_praying ? 'fill-[#FB7185] animate-pulse drop-shadow-[0_0_8px_rgba(251,113,133,0.6)]' : ''} />
   423→                      <span>{prayer.is_praying ? 'I Lifted in Prayer' : 'Pray'}</span>
   424→                      {prayer.praying_count > 0 && (
   425→                        <span className={`text-[10px] px-1.5 py-0.5 rounded-md font-bold ${prayer.is_praying ? 'bg-[#FB7185]/20' : 'bg-white/[0.04]'}`}>
   426→                          {prayer.praying_count}
   427→                        </span>
   428→                      )}
   429→                    </button>
   430→
   431→                    <button
   432→                      onClick={() => toggleComments(prayer._id)}
   433→                      className={`flex items-center gap-1.5 font-bold text-[#64748B] hover:text-white transition-colors px-3 py-1.5 rounded-xl ${
   434→                        isCommentExpanded ? 'bg-white/[0.02] text-white' : ''
   435→                      }`}
   436→                    >
   437→                      <MessageCircle size={15} />
   438→                      <span>Encourage</span>
   439→                      {prayer.comments?.length > 0 && (
   440→                        <span className="text-[10px] bg-white/[0.04] px-1.5 py-0.5 rounded-md font-bold text-slate-400">
   441→                          {prayer.comments.length}
   442→                        </span>
   443→                      )}
   444→                    </button>
   445→                  </div>
   446→
   447→                  {/* Comments/Encouragement Section */}
   448→                  <AnimatePresence>
   449→                    {isCommentExpanded && (
   450→                      <motion.div
   451→                        initial={{ opacity: 0, height: 0 }}
   452→                        animate={{ opacity: 1, height: 'auto' }}
   453→                        exit={{ opacity: 0, height: 0 }}
   454→                        className="overflow-hidden mt-3 pt-3 border-t border-white/[0.04]"
   455→                      >

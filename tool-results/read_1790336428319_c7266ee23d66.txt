   160→def hash_password(password: str) -> str:
   161→    salt = bcrypt.gensalt()
   162→    hashed = bcrypt.hashpw(password.encode("utf-8"), salt)
   163→    return hashed.decode("utf-8")

   165→def verify_password(plain_password: str, hashed_password: str) -> bool:
   166→    return bcrypt.checkpw(plain_password.encode("utf-8"), hashed_password.encode("utf-8"))

   178→async def get_current_user(request: Request) -> dict:
   179→    # Extract token from Authorization header or cookies
   180→    auth_header = request.headers.get("Authorization", "")
   181→    bearer_token = None
   182→    if auth_header.startswith("Bearer "):
   183→        bearer_token = auth_header[7:]
   184→
   185→    # 1. Try session_token first (Google OAuth)
   186→    session_token = request.cookies.get("session_token") or bearer_token
   187→    if session_token:
   188→        session = await db.user_sessions.find_one({"session_token": session_token})
   189→        if session:
   190→            expires_at = session.get("expires_at", datetime.min)
   191→            if isinstance(expires_at, str):
   192→                expires_at = datetime.fromisoformat(expires_at)
   193→            if expires_at.tzinfo is None:
   194→                expires_at = expires_at.replace(tzinfo=timezone.utc)
   195→            if expires_at > datetime.now(timezone.utc):
   196→                user = await db.users.find_one({"_id": ObjectId(session["user_id"])})
   197→                if user:
   198→                    user["_id"] = str(user["_id"])
   199→                    user.pop("password_hash", None)
   200→                    return user
   201→
   202→    # 2. Try Supabase JWT (frontend sends Supabase access_token as Bearer)
   203→    supabase_jwt_secret = os.environ.get("SUPABASE_JWT_SECRET")
   204→    if bearer_token and supabase_jwt_secret:
   205→        try:
   206→            supabase_payload = jwt.decode(bearer_token, supabase_jwt_secret, algorithms=["HS256"])
   207→            # Supabase JWTs have "role" field (anon, authenticated, service_role)
   208→            supabase_sub = supabase_payload.get("sub")
   209→            if supabase_sub and supabase_payload.get("role") in ("authenticated", "anon"):
   210→                # Look up user by supabase_id field in MongoDB
   211→                user = await db.users.find_one({"supabase_id": supabase_sub})
   212→                if user:
   213→                    user["_id"] = str(user["_id"])
   214→                    user.pop("password_hash", None)
   215→                    return user
   216→                # Fallback: look up by email if available in token
   217→                supabase_email = supabase_payload.get("email")
   218→                if supabase_email:
   219→                    user = await db.users.find_one({"email": supabase_email.lower()})
   220→                    if user:
   221→                        # Link supabase_id to this user for future lookups
   222→                        await db.users.update_one(
   223→                            {"_id": ObjectId(user["_id"])},
   224→                            {"$set": {"supabase_id": supabase_sub}}
   225→                        )
   226→                        user["_id"] = str(user["_id"])
   227→                        user.pop("password_hash", None)
   228→                        user["supabase_id"] = supabase_sub
   229→                        return user
   230→        except (jwt.ExpiredSignatureError, jwt.InvalidTokenError):
   231→            pass  # Not a valid Supabase token, try backend JWT next
   232→
   233→    # 3. Try backend's own JWT access_token (email/password auth)
   234→    token = request.cookies.get("access_token") or bearer_token
   235→    if not token:
   236→        raise HTTPException(status_code=401, detail="Not authenticated")
   237→    try:
   238→        payload = jwt.decode(token, get_jwt_secret(), algorithms=[JWT_ALGORITHM])
   239→        if payload.get("type") != "access":
   240→            raise HTTPException(status_code=401, detail="Invalid token type")
   241→        user = await db.users.find_one({"_id": ObjectId(payload["sub"])})
   242→        if not user:
   243→            raise HTTPException(status_code=401, detail="User not found")
   244→        user["_id"] = str(user["_id"])
   245→        user.pop("password_hash", None)
   246→        return user
   247→    except jwt.ExpiredSignatureError:
   248→        raise HTTPException(status_code=401, detail="Token expired")
   249→    except jwt.InvalidTokenError:
   250→        raise HTTPException(status_code=401, detail="Invalid token")

   429→@api_router.post("/auth/login")
   430→async def login(request: LoginRequest, response: Response, req: Request):
   431→    email = request.email.lower()
   432→    identifier = f"{req.client.host}:{email}"
   433→    
   434→    attempt = await db.login_attempts.find_one({"identifier": identifier})
   435→    if attempt and attempt.get("locked_until"):
   436→        locked = attempt["locked_until"]
   437→        if locked.tzinfo is None:
   438→            locked = locked.replace(tzinfo=timezone.utc)
   439→        if locked > datetime.now(timezone.utc):
   440→            raise HTTPException(status_code=429, detail="Too many failed attempts. Try again later.")
   441→    
   442→    user = await db.users.find_one({"email": email})
   443→    if not user or not verify_password(request.password, user["password_hash"]):
   444→        await db.login_attempts.update_one(
   445→            {"identifier": identifier},
   446→            {"$inc": {"count": 1}, "$set": {"last_attempt": datetime.now(timezone.utc)}},
   447→            upsert=True
   448→        )
   449→        current_attempt = await db.login_attempts.find_one({"identifier": identifier})
   450→        if current_attempt and current_attempt.get("count", 0) >= 5:
   451→            await db.login_attempts.update_one(
   452→                {"identifier": identifier},
   453→                {"$set": {"locked_until": datetime.now(timezone.utc) + timedelta(minutes=15)}}
   454→            )
   455→        raise HTTPException(status_code=401, detail="Invalid credentials")
   456→    
   457→    await db.login_attempts.delete_one({"identifier": identifier})
   458→    
   459→    user_id = str(user["_id"])
   460→    access_token = create_access_token(user_id, email)
   461→    refresh_token = create_refresh_token(user_id)
   462→    
   463→    response.set_cookie(key="access_token", value=access_token, httponly=True, secure=False, samesite="lax", max_age=900, path="/")

   627→    if not record:

   653→    if is_video:

   676→async def generate_video_thumbnail(video_data: bytes, user_id: str, file_id: str) -> str:
   677→    """Extract a frame from video at 1s using ffmpeg and upload as thumbnail."""
   678→    with tempfile.TemporaryDirectory() as tmpdir:
   679→        video_path = os.path.join(tmpdir, "input.mp4")
   680→        thumb_path = os.path.join(tmpdir, "thumb.jpg")
   681→        with open(video_path, "wb") as f:
   682→            f.write(video_data)
   683→
   684→        proc = await asyncio.to_thread(
   685→            subprocess.run,
   686→            ["ffmpeg", "-i", video_path, "-ss", "00:00:01", "-vframes", "1", "-vf", "scale=640:-1", "-q:v", "4", thumb_path, "-y"],
   687→            capture_output=True, timeout=30
   688→        )
   689→
   690→        if proc.returncode != 0 or not os.path.exists(thumb_path):
   691→            # Try frame at 0s if 1s fails (short video)
   692→            proc = await asyncio.to_thread(
   693→                subprocess.run,
   694→                ["ffmpeg", "-i", video_path, "-vframes", "1", "-vf", "scale=640:-1", "-q:v", "4", thumb_path, "-y"],
   695→                capture_output=True, timeout=30
   696→            )
   697→
   698→        if os.path.exists(thumb_path):
   699→            with open(thumb_path, "rb") as f:
   700→                thumb_data = f.read()
   701→            thumb_storage_path = f"{APP_NAME}/thumbnails/{user_id}/{file_id}.jpg"
   702→            put_object(thumb_storage_path, thumb_data, "image/jpeg")
   703→            return f"/api/files/{thumb_storage_path}"
   704→
   705→    return ""

   710→
   711→    record = await db.files.find_one({"storage_path": path, "is_deleted": False})
   712→    if not record:
   713→        # Also check thumbnails (no auth record needed)
   714→        if "/thumbnails/" in path:
   715→            try:
   716→                data, content_type = get_object(path)
   717→                return Response(content=data, media_type=content_type, headers={"Cache-Control": "public, max-age=86400"})
   718→            except Exception:
   719→                raise HTTPException(status_code=404, detail="File not found")
   720→        raise HTTPException(status_code=404, detail="File not found")
   721→

   729→        try:
   730→            range_spec = range_header.replace("bytes=", "")
   731→            parts = range_spec.split("-")
   732→            start = int(parts[0]) if parts[0] else 0
   733→            end = int(parts[1]) if parts[1] else total_size - 1
   734→            end = min(end, total_size - 1)
   735→            chunk = data[start:end + 1]
   736→            return Response(
   737→                content=chunk,
   738→                status_code=206,
   739→                media_type=content_type,
   740→                headers={
   741→                    "Content-Range": f"bytes {start}-{end}/{total_size}",
   742→                    "Accept-Ranges": "bytes",
   743→                    "Content-Length": str(len(chunk)),
   744→                    "Cache-Control": "public, max-age=3600",
   745→                }
   746→            )
   747→        except Exception:
   748→            pass
   749→
   750→    return Response(
   751→        content=data,
   752→        media_type=content_type,
   753→        headers={
   754→            "Accept-Ranges": "bytes",
   755→            "Content-Length": str(total_size),
   756→            "Cache-Control": "public, max-age=3600" if content_type.startswith(("image", "video")) else "no-cache",
   757→        }

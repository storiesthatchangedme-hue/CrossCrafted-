   194→// ═══════════════════════════════════════════════════════════════
   195→// Compatibility Proxy: Makes legacy array patterns work with Supabase async ops
   196→// ═════════════════════════════════════════════════════════════════
   197→function createLegacyProxy(table, supabaseModule) {
   198→  if (!DB.isSupabase || !supabaseModule) return null;
   199→
   200→  const handler = {
   201→    get(target, prop) {
   202→      if (prop === 'length') return new Proxy([], { get: () => 0 }); // Can't sync length
   203→      if (typeof prop === 'string' && prop.startsWith('find')) {
   204→        return (...args) => {
   205→          const [queryOrId, callback] = args;
   206→          if (typeof queryOrId === 'function') return queryOrId(callback); // .find(fn)
   207→          // Support both .find(u => ...) and .find(id)
   208→          const isFn = typeof queryOrId === 'function';
   209→          const searchId = isFn ? null : queryOrId;
   210→          return new Proxy([], {
   211→            get: () => searchId ? [] : (isFn ? DB.users.filter(searchId) : []),
   212→            [Symbol.iterator]: async function* () {
   213→              let results;
   214→              if (searchId) {
   215→                results = [DB.users.find(u => u._id === searchId) || DB.users.find(u => u.legacy_id === searchId)].filter(Boolean);
   216→              } else {
   217→                results = isFn ? DB.users.filter(searchId) : [];
   218→              }
   219→              for (const item of results) yield item;
   220→            },
   221→            forEach: () => {}, // Legacy compatibility
   222→            map: (fn) => DB.users.filter(fn), // .find(fn) pattern
   223→            filter: (fn) => DB.users.filter(fn),
   224→            find: (fn) => DB.users.find(fn),
   225→            some: (fn) => DB.users.some(fn),
   226→            every: (fn) => DB.users.every(fn),
   227→            reduce: () => {}, // Not used
   228→            includes: (id) => !!DB.users.find(u => u._id === id || u.legacy_id === id),
   229→            indexOf: (id) => { const u = DB.users.find(u => u._id === id || u.legacy_id === id); return u ? 0 : -1; },
   230→            splice: () => { console.warn('[Proxy] .splice() not supported on Supabase proxy'); },
   231→          });
   232→        };
   233→      }
   234→      // Handle .filter(q), .map(), .sort(), etc.
   235→      return new Proxy([], {
   236→        get: () => [],
   237→        [Symbol.iterator]: async function* () {
   238→          // For .map or direct iteration — just return the current array
   239→          // The server code typically does db.posts.map(...) so DB[table] is the real array
   240→          const arr = DB[table] || [];
   241→          for (const item of arr) yield item;
   242→        },
   243→        filter: (fn) => Array.isArray(DB[table]) ? DB[table].filter(fn) : [],
   244→        find: (fn) => Array.isArray(DB[table]) ? DB[table].find(fn) : null,
   245→        some: (fn) => Array.isArray(DB[table]) ? DB[table].some(fn) : false,
   246→        every: (fn) => Array.isArray(DB[table]) ? DB[table].every(fn) : true,
   247→        includes: (id) => Array.isArray(DB[table]) ? DB[table].some(u => u._id === id || u.legacy_id === id) : false,
   248→        indexOf: (id) => { const arr = DB[table] || []; return arr.findIndex(u => u._id === id || u.legacy_id === id); },
   249→        length: Array.isArray(DB[table]) ? DB[table].length : 0,
   250→        map: (fn) => Array.isArray(DB[table]) ? DB[table].map(fn) : [],
   251→        forEach: (fn) => { if (Array.isArray(DB[table])) DB[table].forEach(fn); },
   252→        sort: (fn) => { if (Array.isArray(DB[table])) DB[table].sort(fn); },
   253→        unshift: (item) => { console.warn(`[Proxy] .unshift() not supported on ${table} — use db.${table}.create()`); },
   254→        push: (item) => { console.warn(`[Proxy] .push() not supported on ${table} — use db.${table}.create()`); },
   255→      };
   256→    },
   257→    set(target, prop, value) {
   258→      if (prop === 'length' || prop === 'prototype') return true;
   259→      return true; // Silently accept all mutations (server handles them)
   260→    },
   261→  };
   262→
   263→  return new Proxy(handler, handler);
   264→}

   840→// --- Authentication Middleware ---
   841→// Works with both Supabase and legacy in-memory DB
   842→async function authenticateToken(req, res, next) {
   843→  const authHeader = req.headers['authorization'];
   844→  const token = (authHeader && authHeader.split(' ')[1]) || req.cookies.access_token || req.cookies.refresh_token;
   845→
   846→  if (!token) {
   847→    return res.status(401).json({ detail: 'Authentication token missing or invalid' });
   848→  }
   849→
   850→  jwt.verify(token, JWT_SECRET, async (err, decoded) => {
   851→    if (err) {
   852→      return res.status(401).json({ detail: 'Invalid or expired session' });
   853→    }
   854→
   855→    let user;
   856→    if (DB.isSupabase) {
   857→      user = await db.users.findById(decoded.userId);
   858→    } else {
   859→      user = DB.users.find(u => u._id === decoded.userId);
   860→    }
   861→
   862→    if (!user) {
   863→      return res.status(401).json({ detail: 'User account not found' });
   864→    }
   865→    if (user.status === 'rejected') {
   866→      return res.status(403).json({ detail: 'Your account registration was not approved.' });
   867→    }
   868→    req.user = user;
   869→    next();
   870→  });
   871→}

   873→// Optional Auth Middleware (doesn't fail if no token)
   874→async function optionalAuthenticateToken(req, res, next) {
   875→  const authHeader = req.headers['authorization'];
   876→  const token = (authHeader && authHeader.split(' ')[1]) || req.cookies.access_token || req.cookies.refresh_token;
   877→
   878→  if (!token) {
   879→    return next();
   880→  }
   881→
   882→  jwt.verify(token, JWT_SECRET, async (err, decoded) => {
   883→    if (!err) {
   884→      let user;
   885→      if (DB.isSupabase) {
   886→        user = await db.users.findById(decoded.userId);
   887→      } else {
   888→        user = DB.users.find(u => u._id === decoded.userId);
   889→      }
   890→      if (user && user.status !== 'rejected') {
   891→        req.user = user;
   892→      }
   893→    }
   894→    next();
   895→  });
   896→}

  1022→
  1023→  if (!code) {
  1024→    return res.status(400).send('Authorization code is missing.');
  1025→  }
  1026→
  1027→  try {
  1028→    const clientId = process.env.GOOGLE_CLIENT_ID;
  1029→    const clientSecret = process.env.GOOGLE_CLIENT_SECRET;
  1030→    const redirectUri = getRedirectUri(req);
